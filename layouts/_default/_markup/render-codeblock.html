{{- $lang := lower (.Type | default "") -}} {{- $code := trim .Inner "\n\r" -}} {{- /* Languages that should be presented as a terminal session */ -}} {{- $shells := slice "bash" "sh" "shell" "zsh" "fish" "console" "terminal" "shell-session" "shellsession" "session" "powershell" "pwsh" "ps1" "bat" "batch" "cmd" "gdb" -}} {{- $isShell := in $shells $lang -}} {{- /* Chroma has no lexer for these, but they are still shell-adjacent config */ -}} {{- $known := slice "apparmor" "nftables" "iptables" -}} {{- $hlLang := $lang -}} {{- if in $known $lang -}}{{- $hlLang = "text" -}}{{- end -}} {{- $rendered := "" -}} {{- if $hlLang -}} {{- $rendered = highlight $code $hlLang (.Options | default dict) -}} {{- else -}} {{- $rendered = printf "
%s
" (htmlEscape $code) | safeHTML -}} {{- end -}} {{- if $isShell -}} {{- $shell := $lang -}} {{- if in (slice "pwsh" "ps1") $lang -}}{{- $shell = "powershell" -}} {{- else if in (slice "bat" "batch") $lang -}}{{- $shell = "cmd" -}} {{- else if in (slice "console" "terminal" "session" "shell-session" "shellsession" "shell") $lang -}}{{- $shell = "sh" -}} {{- end -}} {{- $title := printf "%s — mahmoud@portfolio" $shell -}} {{- if eq $shell "gdb" -}}{{- $title = "gdb — debug session" -}}{{- end -}} {{- $interactive := ne $shell "gdb" -}} {{- $promptSym := "$" -}}{{- if eq $shell "zsh" -}}{{- $promptSym = "%" -}}{{- end -}} {{- $promptStr := printf "mahmoud@portfolio ~ %s" $promptSym -}}
{{ $title }}
{{ $rendered }}
{{- if $interactive }}
{{ $promptStr }}
{{- end }}
{{- else -}} {{ $rendered }} {{- end -}}
